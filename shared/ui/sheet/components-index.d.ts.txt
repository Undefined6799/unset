import type * as React from "react";
import type { ReactNode, MouseEvent } from "react";

export interface ButtonProps {
  /** The label, set in the `label` style, uppercased by the component. One to three words. */
  children: ReactNode;
  /** Renders an <a> instead of a <button>. */
  href?: string;
  /** Native button type; defaults to "button". */
  type?: "button" | "submit" | "reset";
  disabled?: boolean;
  onClick?: (event: MouseEvent<HTMLElement>) => void;
  className?: string;
}

/** Solid `ink` button, no border, square with the bottom-right corner cut off at `cut`. */
export declare function Button(props: ButtonProps): JSX.Element;

export interface MarkProps {
  /** Rendered width in px; height follows the mark's 828:220 ratio. Default 72. */
  width?: number;
  /** Accessible name. Default "unset.sh". */
  title?: string;
  className?: string;
}
/** The three-fisheye mark at its exact source geometry, filled with currentColor (`ink` by default). */
export declare function Mark(props: MarkProps): JSX.Element;

export interface CommandBlockProps {
  /** The command exactly as the user should paste it. */
  command: string;
  /** Prompt glyph shown before the command, not copied. Default "$"; pass "" for none. */
  prompt?: string;
  /** Optional output lines. Lines starting "[ok]", "[err]" or "[info]" get a coloured tag; lines starting "#" are muted. */
  output?: string[];
  /** Show the copy control. Default true. */
  copyable?: boolean;
  className?: string;
}
/** A shell command on `surface-raised` with a copy control, and optional output below. */
export declare function CommandBlock(props: CommandBlockProps): JSX.Element;

export interface InputProps extends Omit<React.InputHTMLAttributes<HTMLInputElement>, "className"> {
  /** Label above the field, uppercased. */
  label?: string;
  /** Helper text under the field. */
  hint?: string;
  /** Error text; turns the border `danger` and prints "[err] <error>" under the field. */
  error?: string;
  className?: string;
}
/** Single-line text field with the button's 2px corner. */
export declare function Input(props: InputProps): JSX.Element;

export interface LinkProps extends Omit<React.AnchorHTMLAttributes<HTMLAnchorElement>, "className"> {
  /** inline: underlined `link` text inside a sentence (default). standalone: an uppercase `label` link on its own line, usually ending in "→". */
  variant?: "inline" | "standalone";
  children: ReactNode;
  className?: string;
}
export declare function Link(props: LinkProps): JSX.Element;

export interface TagProps {
  /** ok: `success`. err: `danger`. info: `link`. plain: `ink-muted` (default). */
  status?: "ok" | "err" | "info" | "plain";
  /** The word inside the brackets; defaults to the status name. */
  children?: ReactNode;
  className?: string;
}
/** Bracketed mono text, "[err]"; no fill, no border. */
export declare function Tag(props: TagProps): JSX.Element;

export interface CalloutProps {
  /** info (default, glyph "i"), danger ("!"), success ("+"), note ("*"). */
  tone?: "info" | "danger" | "success" | "note";
  /** Override the glyph: one Unicode character that JetBrains Mono draws (never an emoji). */
  glyph?: string;
  title?: ReactNode;
  children?: ReactNode;
  className?: string;
}
/** A notice on `surface-raised` led by one coloured Unicode glyph. */
export declare function Callout(props: CalloutProps): JSX.Element;

export interface HeaderProps {
  /** Nav links; mark one `current`. */
  nav?: { label: string; href: string; current?: boolean }[];
  /** One Button at the end of the nav. */
  action?: { label: string; href?: string; onClick?: (event: MouseEvent<HTMLElement>) => void };
  /** Where the mark links. Default "/". */
  homeHref?: string;
  className?: string;
}
/** Site header: the mark on the left, uppercase nav and one Button on the right, over a `line` rule. */
export declare function Header(props: HeaderProps): JSX.Element;

export interface CardProps {
  /** Small mono line above the title: an index ("01"), a path or a date. */
  eyebrow?: ReactNode;
  title?: ReactNode;
  /** One or two sentences in `ink-muted`. */
  children?: ReactNode;
  /** Optional standalone link at the bottom, e.g. { label: "Read more →", href }. */
  action?: { label: string; href: string };
  className?: string;
}
/** Level 2: solid `surface-card` with `shadow-2`, 2px corners, no border. Solid approved 2026-10-04. */
export declare function Card(props: CardProps): JSX.Element;

export declare function SectionHeading(props: { title: ReactNode; index?: string; className?: string }): JSX.Element;

export interface ModalProps {
  open: boolean;
  onClose?: () => void;
