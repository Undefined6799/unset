    var cls = cx("us-avatar", "us-avatar--" + size, props.className);
    var alt = props.alt != null ? props.alt : "";
    if (props.src) return h("span", { className: cls }, h("img", { className: "us-avatar__img", src: props.src, alt: alt, width: size, height: size, loading: "lazy", decoding: "async" }));
    return h("span", alt ? { className: cls, role: "img", "aria-label": alt } : { className: cls, "aria-hidden": "true" }, firstChar(props.name));
  }

  function Switch(props) {
    var st = R.useState(!!props.defaultChecked), own = st[0], setOwn = st[1];
    var on = props.checked != null ? props.checked : own;
    return h("label", { className: cx("us-switch", props.disabled && "is-disabled", props.className) },
      h("span", { className: "us-switch__label" }, props.label),
      h("input", { type: "checkbox", role: "switch", className: "us-visually-hidden", name: props.name, value: props.value, checked: on, disabled: props.disabled,
        onChange: function (e) { if (props.checked == null) setOwn(e.target.checked); if (props.onChange) props.onChange(e); } }),
      h("span", { className: "us-switch__track", "aria-hidden": "true" },
        h("span", { className: "is-on" }, "[──◉] on "),
        h("span", { className: "is-off" }, "[○──] off")));
  }

  function Toast(props) {
    var tone = props.tone || "ok";
    var word = { ok: "ok", err: "err", info: "info" }[tone] || "ok";
    var box = props.children ? h("div", { className: "us-toast" },
      h(Tag, { status: word }),
      h("span", { className: "us-toast__text" }, props.children),
      props.onClose ? h("button", { type: "button", className: "us-toast__close", onClick: props.onClose },
        h(Icon, { name: "close", size: 16 }), h("span", { className: "us-visually-hidden" }, props.closeLabel || "Dismiss")) : null) : null;
    return h("div", { className: cx("us-toast-region", props.inline && "us-toast-region--inline", props.className),
      role: tone === "err" ? "alert" : "status", "aria-live": tone === "err" ? "assertive" : "polite" }, box);
  }

  function SkipLink(props) {
    return h("a", { className: cx("us-skip", props.shown && "is-shown", props.className), href: props.href || "#main" }, props.children || "Skip to content");
  }

  var RATIOS = { "1:1": "1 / 1", "4:5": "4 / 5", "9:16": "9 / 16", "16:9": "16 / 9", "3:1": "3 / 1" };
  function MediaFrame(props) {
    var ratio = RATIOS[props.ratio] || RATIOS["1:1"];
    var box = h("div", { className: "us-frame__box", style: { aspectRatio: ratio } },
      props.src ? h("img", { className: "us-frame__img", src: props.src, alt: props.alt, loading: "lazy", decoding: "async" })
                : h("span", { className: "us-frame__empty" }, props.emptyText || "[no image]"));
    return h("figure", { className: cx("us-frame", props.className), style: props.width ? { width: props.width } : null },
      box,
      props.caption ? h("figcaption", { className: "us-frame__caption" }, props.caption) : null);
  }

  function pageList(page, pages) {
    var out = [], want = {}, i;
    [1, pages, page - 1, page, page + 1].forEach(function (n) { if (n >= 1 && n <= pages) want[n] = true; });
    var prev = 0;
    for (i = 1; i <= pages; i++) {
      if (!want[i]) continue;
      if (prev && i - prev > 1) out.push("gap" + i);
      out.push(i); prev = i;
    }
    return out;
  }
  function Pagination(props) {
    var label = props.label || "Pages";
    function arrow(name) { return h(Icon, { name: name, size: 16 }); }
    function item(href, text, extra, key) {
      var inner = h("span", null, text);
      if (!href) return h("li", { key: key }, h("span", { className: cx("us-pages__item", "is-off", extra) }, inner));
      return h("li", { key: key }, h("a", { className: cx("us-pages__item", extra), href: href }, inner));
    }
    if (props.pages == null) {
      return h("nav", { className: cx("us-pages", props.className), "aria-label": label },
        h("ul", { className: "us-pages__list us-pages__list--cursor" },
          item(props.newer, h(R.Fragment, null, arrow("back"), "newer"), null, "newer"),
          item(props.older, h(R.Fragment, null, "older", arrow("next")), null, "older")));
    }
    var page = props.page, pages = props.pages, hrefFor = props.hrefFor || function (n) { return "?page=" + n; };
    return h("nav", { className: cx("us-pages", props.className), "aria-label": label },
      h("ul", { className: "us-pages__list" },
        item(page > 1 ? hrefFor(page - 1) : null, h(R.Fragment, null, arrow("back"), "prev"), null, "prev"),
        pageList(page, pages).map(function (n) {
          if (typeof n === "string") return h("li", { key: n, className: "us-pages__gap", "aria-hidden": "true" }, "…");
          if (n === page) return h("li", { key: n }, h("a", { className: "us-pages__item us-pages__num is-current", href: hrefFor(n), "aria-current": "page" }, h("span", null, String(n))));
          return item(hrefFor(n), String(n), "us-pages__num", n);
        }),
        h("li", { key: "count", className: "us-pages__count" }, page + " / " + pages),
        item(page < pages ? hrefFor(page + 1) : null, h(R.Fragment, null, "next", arrow("next")), null, "next")));
  }

  function DescriptionList(props) {
    return h("dl", { className: cx("us-dl", props.className) },
      (props.items || []).map(function (it, i) {
        var empty = it.value == null || it.value === "";
        return h("div", { key: it.key || i, className: "us-dl__row" },
          h("dt", { className: "us-dl__label" }, it.label),
          h("dd", { className: cx("us-dl__value", it.mono && "is-mono") }, empty ? "—" : it.value));
      }));
  }

  function NewPosts(props) {
    var n = props.count || 0;
    var text = props.children || (n === 1 ? "1 new post" : (n > 99 ? "99+ new posts" : (n > 0 ? n + " new posts" : "new posts")));
    return h("div", { className: cx("us-newposts", props.inline && "us-newposts--inline", props.hidden && "is-hidden", props.className) },
      h("span", { className: "us-newposts__text" }, h(Tag, { status: "info" }), " " + text),
      h("a", { className: "us-newposts__show", href: props.href || "?", onClick: props.onClick },
        h("span", null, "show"), h(Icon, { name: "up", size: 16 })));
  }

  function FeedMore(props) {
    var state = props.state || "idle";
    var body;
    if (state === "loading") body = h(Spinner, { label: props.loadingText || "loading older posts" });
    else if (state === "end") body = h("span", { className: "us-feedmore__end" }, h(Tag, { status: "plain" }, "end"), " " + (props.endText || "you're all caught up"));
    else if (state === "error") body = h("span", { className: "us-feedmore__err" }, h(Tag, { status: "err" }),
      " " + (props.errorText || "couldn't load more") + " ",
      h("a", { className: "us-feedmore__link", href: props.older || "?", onClick: props.onRetry }, "try again"));
    else body = h("a", { className: "us-feedmore__link", href: props.older }, h("span", null, "older posts"), h(Icon, { name: "down", size: 16 }));
    return h("div", { className: cx("us-feedmore", props.className), "aria-live": "polite" }, body);
  }

  window.Unset = { Button: Button, CommandBlock: CommandBlock, Input: Input, Link: Link, Tag: Tag, Callout: Callout, Card: Card, Header: Header, Mark: Mark, SectionHeading: SectionHeading, Tabs: Tabs, CodeBlock: CodeBlock, Table: Table, Checkbox: Checkbox, RadioGroup: RadioGroup, Textarea: Textarea, Select: Select, Kbd: Kbd, Progress: Progress, Spinner: Spinner, AsciiBackground: AsciiBackground, Footer: Footer, Modal: Modal, Icon: Icon, Avatar: Avatar, Switch: Switch, Toast: Toast, SkipLink: SkipLink, MediaFrame: MediaFrame, Pagination: Pagination, DescriptionList: DescriptionList, NewPosts: NewPosts, FeedMore: FeedMore };
